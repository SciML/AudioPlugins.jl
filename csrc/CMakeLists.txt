cmake_minimum_required(VERSION 3.16)
project(AudioPluginsLive LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(AP_LIVE_DEVICE "Include native audio devices" ON)
option(AP_LIVE_LV2 "Include LV2 (requires lilv)" ON)
option(AP_LIVE_VST3 "Include VST3 (requires built SDK)" ON)
set(VST3_SDK_ROOT "" CACHE PATH "VST3 SDK include directory")
set(VST3_SDK_LIBDIR "" CACHE PATH "VST3 SDK static library directory")
add_library(audioplugins_live SHARED clap_live.c)
target_include_directories(audioplugins_live PRIVATE vendor)
find_package(Threads REQUIRED)
target_link_libraries(audioplugins_live PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
if(WIN32)
  set_target_properties(audioplugins_live PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
  target_link_libraries(audioplugins_live PRIVATE avrt ole32 uuid user32 winmm shlwapi shell32)
elseif(APPLE)
  target_link_libraries(audioplugins_live PRIVATE "-framework CoreFoundation" "-framework CoreAudio" "-framework AudioToolbox")
else()
  target_link_libraries(audioplugins_live PRIVATE m)
endif()
if(AP_LIVE_DEVICE)
  target_sources(audioplugins_live PRIVATE live_device.c)
  target_compile_definitions(audioplugins_live PRIVATE AP_LIVE_WITH_DEVICE)
endif()
if(AP_LIVE_LV2)
  find_path(LILV_INCLUDE_DIR lilv/lilv.h PATH_SUFFIXES lilv-0 REQUIRED)
  find_library(LILV_LIBRARY NAMES lilv-0 REQUIRED)
  target_sources(audioplugins_live PRIVATE lv2_live.c)
  target_include_directories(audioplugins_live PRIVATE ${LILV_INCLUDE_DIR})
  target_link_libraries(audioplugins_live PRIVATE ${LILV_LIBRARY})
endif()
if(AP_LIVE_VST3)
  set(HOSTING ${VST3_SDK_ROOT}/public.sdk/source/vst/hosting)
  if(APPLE)
    enable_language(OBJCXX)
    set(MODULE ${HOSTING}/module_mac.mm)
    set_source_files_properties(${MODULE} PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
    target_link_libraries(audioplugins_live PRIVATE "-framework Cocoa")
  elseif(WIN32)
    set(MODULE ${HOSTING}/module_win32.cpp)
  else()
    set(MODULE ${HOSTING}/module_linux.cpp)
  endif()
  target_sources(audioplugins_live PRIVATE vst3_live.cpp ${HOSTING}/plugprovider.cpp
    ${VST3_SDK_ROOT}/public.sdk/source/vst/utility/stringconvert.cpp
    ${VST3_SDK_ROOT}/public.sdk/source/common/commonstringconvert.cpp ${MODULE})
  target_include_directories(audioplugins_live PRIVATE ${VST3_SDK_ROOT})
  target_compile_definitions(audioplugins_live PRIVATE RELEASE=1)
  foreach(NAME sdk_hosting sdk_common sdk base pluginterfaces)
    find_library(VST3_${NAME} NAMES ${NAME} PATHS ${VST3_SDK_LIBDIR} NO_DEFAULT_PATH REQUIRED)
    target_link_libraries(audioplugins_live PRIVATE ${VST3_${NAME}})
  endforeach()
endif()
install(TARGETS audioplugins_live LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(FILES clap_live.h live_device.h DESTINATION include/audioplugins)
